def merge_sorted_lists(list1, list2):
    """
    Merges two sorted lists into a single sorted list.
    
    Args:
        list1 (list): First sorted list
        list2 (list): Second sorted list
    
    Returns:
        list: Merged sorted list
    """
    merged = []
    i, j = 0, 0
    
    # Compare elements from both lists and add the smaller one
    while i < len(list1) and j < len(list2):
        if list1[i] <= list2[j]:
            merged.append(list1[i])
            i += 1
        else:
            merged.append(list2[j])
            j += 1
    
    # Add remaining elements from list1, if any
    while i < len(list1):
        merged.append(list1[i])
        i += 1
    
    # Add remaining elements from list2, if any
    while j < len(list2):
        merged.append(list2[j])
        j += 1
    
    return merged


# Unit tests
def test_merge_sorted_lists():
    # Test case 1: Normal merging of two sorted lists
    list1 = [1, 3, 5]
    list2 = [2, 4, 6]
    expected = [1, 2, 3, 4, 5, 6]
    assert merge_sorted_lists(list1, list2) == expected, "Test 1 failed"
    
    # Test case 2: One list is empty
    list1 = [1, 2, 3]
    list2 = []
    expected = [1, 2, 3]
    assert merge_sorted_lists(list1, list2) == expected, "Test 2 failed"
    
    # Test case 3: Both lists are empty
    list1 = []
    list2 = []
    expected = []
    assert merge_sorted_lists(list1, list2) == expected, "Test 3 failed"
    
    print("All tests passed!")


# Run the tests
test_merge_sorted_lists()
